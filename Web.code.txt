async function toggleCamera() {
  if (camActive) {
    stopCamera();
  } else {
    await startCamera();
  }
}

async function startCamera() {
  try {
    const btn = document.getElementById('camBtn');
    btn.innerHTML = '<div class="spinner"></div> جاري التشغيل...';
    btn.disabled = true;

    mediaStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user', width: 640, height: 480 }, audio: false });
    const video = document.getElementById('camVideo');
    video.srcObject = mediaStream;
    video.style.display = 'block';
    document.getElementById('camPlaceholder').style.display = 'none';
    camActive = true;

    document.getElementById('camLiveDot').style.background = '#ef4444';
    document.getElementById('camLiveLabel').textContent = 'الكاميرا — تحليل الإشارة 🟢';
    document.getElementById('camStatus').style.display = 'block';
    document.getElementById('camStatus').textContent = '✅ الكاميرا تعمل — يتم تحليل الإشارات';

    btn.className = 'ctrl-btn ctrl-btn-danger';
    btn.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="16" height="16"><path d="M23 7l-7 5 7 5V7z"/><rect x="1" y="5" width="15" height="14" rx="2"/></svg> إيقاف الكاميرا';
    btn.disabled = false;

    updateStatus('الكاميرا تعمل', 'connected');
    startSignDetection();
    toast('✅ الكاميرا تعمل — جاهز لكشف الإشارة');
  } catch (e) {
    document.getElementById('camBtn').className = 'ctrl-btn ctrl-btn-primary';
    document.getElementById('camBtn').innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="16" height="16"><path d="M23 7l-7 5 7 5V7z"/><rect x="1" y="5" width="15" height="14" rx="2"/></svg> تشغيل الكاميرا';
    document.getElementById('camBtn').disabled = false;
    toast('⚠️ تعذّر الوصول للكاميرا — تحقق من الصلاحيات');
  }
}

function stopCamera() {
  if (mediaStream) { mediaStream.getTracks().forEach(t => t.stop()); mediaStream = null; }
  const video = document.getElementById('camVideo');
  video.style.display = 'none';
  document.getElementById('camPlaceholder').style.display = 'flex';
  document.getElementById('camStatus').style.display = 'none';
  document.getElementById('camLiveDot').style.background = '#888';

  document.getElementById('detectionBox').classList.remove('visible');
  document.getElementById('signTranslation').textContent = 'في انتظار الإشارة...';

  camActive = false;
  const btn = document.getElementById('camBtn');
  btn.className = 'ctrl-btn ctrl-btn-primary';
  btn.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="16" height="16"><path d="M23 7l-7 5 7 5V7z"/><rect x="1" y="5" width="15" height="14" rx="2"/></svg> تشغيل الكاميرا';
  updateStatus('في انتظار البدء', 'disconnected');
}

// ══════════════════════════════════════
//  SIMULATED SIGN DETECTION
// (In production: replace with ML model like MediaPipe + custom classifier)

const signPhrases = [
  'أحتاج مساعدة','شكراً جزيلاً','لم أفهم','أعد من فضلك',
  'نعم، فهمت','هل يمكنك التوضيح؟','ما معنى هذا؟','أستاذ لدي سؤال',
  'صباح الخير','كيف حالك؟','أنا بخير','هل الامتحان غداً؟',
  'الدرس واضح','أريد الإجابة','شكراً للشرح'
];

let signPhraseIdx = 0;
function startSignDetection() {
  animateBars();
  signInterval = setInterval(() => {
    if (!camActive) return;
    // Simulate detection box
    const box = document.getElementById('detectionBox');
    const video = document.getElementById('camVideo');
    const vw = video.clientWidth, vh = video.clientHeight;
    const bw = 140 + Math.random()*60, bh = 160 + Math.random()*60;
    const bx = 20 + Math.random()*(vw - bw - 40);
    const by = 20 + Math.random()*(vh - bh - 40);
    box.style.width = bw+'px'; box.style.height = bh+'px';
    box.style.left = bx+'px'; box.style.top = by+'px';
    box.classList.add('visible');

    // Simulate translation
    const phrase = signPhrases[signPhraseIdx % signPhrases.length];
    signPhraseIdx++;
    document.getElementById('detectionLabel').textContent = 'إشارة محددة';
    document.getElementById('signTranslation').textContent = phrase;

    // Auto-add to chat
    addMessage(phrase, 'them', 'sign');
  }, 7000);
}

function animateBars() {
  const bars = document.querySelectorAll('#signBars .wave-bar');
  setInterval(() => {
    if (!camActive) { bars.forEach(b => b.style.transform = 'scaleY(.3)'); return; }
    bars.forEach(b => { b.style.transform = `scaleY(${.2 + Math.random()*.9})`; });
  }, 120);
}

// ══════════════════════════════════════
//  SPEECH RECOGNITION
// ══════════════════════════════════════
function toggleSpeechRecognition() {
  if (speechActive) { stopSpeech(); } else { startSpeech(); }
}

function startSpeech() {
  const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SpeechRecognition) { toast('⚠️ المتصفح لا يدعم التعرف على الصوت — جرب Chrome'); return; }

  recognition = new SpeechRecognition();
  recognition.lang = document.getElementById('speechLang')?.value || 'ar-SA';
  recognition.continuous = true;
  recognition.interimResults = true;

  recognition.onstart = () => {
    speechActive = true;
    updateStatus('يستمع للصوت...', 'recording-status');
    const btn = document.getElementById('speechBtn');
    btn.className = 'ctrl-btn ctrl-btn-danger';
    btn.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="16" height="16"><path d="M12 2a3 3 0 013 3v7a3 3 0 01-6 0V5a3 3 0 013-3z"/><path d="M19 10v2a7 7 0 01-14 0v-2M12 19v4"/></svg> إيقاف الاستماع';
    toast('🎙️ يستمع... تكلم الآن');
  };

  recognition.onresult = (e) => {
    let interim = '', final = '';
    for (let i = e.resultIndex; i < e.results.length; i++) {
      if (e.results[i].isFinal) final += e.results[i][0].transcript;
      else interim += e.results[i][0].transcript;
    }
    if (interim) document.getElementById('chatInput').value = interim;
    if (final) {
      document.getElementById('chatInput').value = final;
      const autoSend = document.getElementById('toggleAutoSend')?.classList.contains('on');
      if (autoSend) { sendMessage(); } else { document.getElementById('chatInput').focus(); }
    }
  };

  recognition.onerror = (e) => {
    if (e.error !== 'no-speech') toast('⚠️ خطأ في التعرف على الصوت: ' + e.error);
  };

  recognition.onend = () => {
    if (speechActive) recognition.start(); // keep alive
  };

  recognition.start();
}

function stopSpeech() {
  speechActive = false;
  if (recognition) { recognition.onend = null; recognition.stop(); recognition = null; }
  updateStatus(camActive ? 'الكاميرا تعمل' : 'في انتظار البدء', camActive ? 'connected' : 'disconnected');
  const btn = document.getElementById('speechBtn');
  btn.className = 'ctrl-btn ctrl-btn-default';
  btn.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" width="16" height="16"><path d="M12 2a3 3 0 013 3v7a3 3 0 01-6 0V5a3 3 0 013-3z"/><path d="M19 10v2a7 7 0 01-14 0v-2M12 19v4"/></svg> تحويل الصوت لنص';
  toast('⏹️ توقف الاستماع');
}

function addMessage(text, side, type) {
  if (!text.trim()) return;
  const time = new Date().toLocaleTimeString('ar', { hour: '2-digit', minute: '2-digit' });
  const typeLabel = { sign: 'إشارة', voice: 'صوت', text: 'نص' };
  const typeClass = { sign: 'type-sign', voice: 'type-voice', text: 'type-text' };

  const msg = { text, side, type, time };
  chatLog.push(msg);

  const el = document.createElement('div');
  el.className = 'msg ' + (side === 'me' ? 'me' : 'them');
  el.innerHTML = `
    <div class="msg-bubble">${text}</div>
    <div class="msg-meta">
      <span>${side === 'me' ? (role === 'student' ? 'أنت' : 'الدكتور') : 'الطرف الآخر'}</span>
      <span class="msg-type ${typeClass[type] || 'type-text'}">${typeLabel[type] || 'نص'}</span>
      <span>${time}</span>
    </div>`;
  const msgs = document.getElementById('messages');
  msgs.appendChild(el);
  msgs.scrollTop = msgs.scrollHeight;

  // TTS for incoming messages if enabled
  const tts = document.getElementById('toggleTTS')?.classList.contains('on');
  if (side === 'them' && tts) speak(text);
}

function sendMessage() {
  const inp = document.getElementById('chatInput');
  const text = inp.value.trim();
  if (!text) return;
  addMessage(text, 'me', speechActive ? 'voice' : 'text');
  inp.value = '';
  // simulate reply
  setTimeout(() => {
    const replies = [
      'شكراً لك، هل هناك أي استفسار آخر؟',
      'مفهوم! سأشرح هذه النقطة بتفصيل أكبر',
      'سؤال ممتاز! الإجابة هي...',
      'بالتأكيد، دعني أوضح هذا',
      'أحسنت! هذا صحيح تماماً'
    ];
    addMessage(replies[Math.floor(Math.random()*replies.length)], 'them', 'text');
  }, 1200 + Math.random()*800);
}

function clearChat() {
  document.getElementById('messages').innerHTML = '';
  chatLog = [];
  toast('تم مسح المحادثة');
}

function speakLastMessage() {
  const last = chatLog.filter(m => m.side === 'them').pop();
  if (last) speak(last.text); else toast('لا توجد رسائل لقراءتها');
}

function speak(text) {
  if (!window.speechSynthesis) return;
  speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(text);
  u.lang = 'ar-SA'; u.rate = .9; u.pitch = 1;
  speechSynthesis.speak(u);
}

function saveConversation() {
  if (!chatLog.length) { toast('لا توجد رسائل لحفظها'); return; }
  const session = {
    id: Date.now(),
    date: new Date().toLocaleDateString('ar-SA'),
    time: new Date().toLocaleTimeString('ar-SA', { hour: '2-digit', minute: '2-digit' }),
    role,
    messages: [...chatLog],
    title: `جلسة ${role === 'student' ? 'طالب' : 'دكتور'} — ${new Date().toLocaleDateString('ar-SA')}`
  };
  sessions.unshift(session);
  localStorage.setItem('nutq_sessions', JSON.stringify(sessions));
  toast('✅ تم حفظ الجلسة');
}

}

    </div>`;

function answerQuiz(chosen, correct, btn) {
  if (quizAnswered) return;
  quizAnswered = true;
  const isCorrect = chosen === correct;
  btn.classList.add(isCorrect ? 'correct' : 'wrong');
  if (!isCorrect) {
    document.querySelectorAll('.quiz-opt').forEach(b => { if (b.textContent === correct) b.classList.add('correct'); });
  }
  if (isCorrect) quizScore++;
  toast(isCorrect ? '✅ إجابة صحيحة!' : `❌ الإجابة الصحيحة: ${correct}`);
  setTimeout(() => { quizIndex++; renderQuiz(); }, 1600);
}
